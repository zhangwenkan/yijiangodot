#version 450
// Source M_水花 PS50198 + UE4 DistortionCommon.ush. No main-pass soft-contact fade.
layout(location=0) in vec2 uv;
layout(location=1) flat in vec4 particle;
layout(location=2) flat in float speed_cm_s;
layout(location=3) in float surface_depth_cm;
layout(location=0) out vec4 encoded_offset;
layout(set=0,binding=1,std140) uniform Frame {
    mat4 projection;
    mat4 inverse_projection;
    mat4 world_to_view;
    vec4 render_and_output_size;
    vec4 source;
} frame;
layout(set=0,binding=2) uniform sampler2D color_a;
layout(set=0,binding=3) uniform sampler2D color_b;
layout(set=0,binding=4) uniform sampler2D normal_a;
layout(set=0,binding=5) uniform sampler2D normal_b;
layout(set=0,binding=6) uniform sampler2D opaque_depth;
vec3 unpack_bc5(vec2 rg) {
    vec2 xy=rg*2.0-1.0;
    return vec3(xy,sqrt(max(1.0-dot(xy,xy),0.0)));
}
float linear_depth_cm(vec2 screen_uv,float device_z) {
    vec4 v=frame.inverse_projection*vec4(screen_uv*2.0-1.0,device_z,1.0);
    return abs(v.w)<1e-12 ? 1e30 : -v.z/v.w*100.0;
}
void main() {
    vec2 size=frame.render_and_output_size.xy;
    vec2 screen_uv=gl_FragCoord.xy/size;
    // Exact reversed-Z depth test against the opaque scene. Transparent ocean never writes it.
    if (gl_FragCoord.z<textureLod(opaque_depth,screen_uv,0.0).r) discard;
    float blend=clamp(speed_cm_s/frame.source.x,0.0,1.0);
    vec3 n=mix(unpack_bc5(texture(normal_a,uv,frame.source.z).rg),
               unpack_bc5(texture(normal_b,uv,frame.source.z).rg),blend);
    // +U follows the billboard right; +V follows its down direction, as in UE SpriteVF.
    // The source/current Z axes differ in handedness, but XY and normal length are invariant.
    vec3 view_normal=normalize(vec3(particle.x*n.x+particle.y*n.y,
                                   particle.y*n.x-particle.x*n.y,n.z));
    float opacity=mix(texture(color_a,uv,frame.source.z).r,
                      texture(color_b,uv,frame.source.z).r,blend)*particle.z;
    vec2 offset=(particle.w-1.0)*opacity*view_normal.xy;
    // Godot supplies a whole-view buffer: source ViewSize/BufferSize is exactly (1,1).
    if (dot(offset,offset)<0.00001) discard;
    float inv_tan_half_horizontal_fov=abs(frame.projection[0][0]);
    vec2 output_size=frame.render_and_output_size.zw;
    offset *= vec2(inv_tan_half_horizontal_fov,
                   inv_tan_half_horizontal_fov*output_size.x/output_size.y);
    offset *= output_size*vec2(0.00023,-0.00023);
    vec2 distorted_uv=screen_uv+offset;
    float distorted_depth=linear_depth_cm(distorted_uv,textureLod(opaque_depth,distorted_uv,0.0).r);
    float bias=-frame.source.y;
    float range=clamp(abs(bias*0.5),0.0,50.0);
    offset *= clamp((distorted_depth-surface_depth_cm)/max(1.0,range)+bias,0.0,1.0);
    offset *= 4.0;
    // Attachment is RGBA8 UNORM; ONE+ONE for RGB AND alpha, with native quantization/saturation.
    encoded_offset=vec4(max(offset,vec2(0.0)),max(-offset,vec2(0.0)));
}
