#version 450
// Rasterize the existing Blender-authored mesh, with the current splash main-pass billboard.
layout(location=0) in vec3 source_position;
layout(location=1) in vec2 source_uv;
struct Particle { vec4 center_angle; vec4 scale_alpha; vec4 pivot_dynamic_speed; };
layout(set=0,binding=0,std430) readonly buffer ParticleBuffer { Particle particles[]; };
layout(set=0,binding=1,std140) uniform Frame {
    mat4 projection;
    mat4 inverse_projection;
    mat4 world_to_view;
    vec4 render_and_output_size;
    vec4 source; // ParticleSpeed, RefractionDepthBias, native mip bias, main-pass angle sign
} frame;
layout(location=0) out vec2 uv;
layout(location=1) flat out vec4 particle; // cos(angle),sin(angle),source alpha,Dynamic.B
layout(location=2) flat out float speed_cm_s;
layout(location=3) out float surface_depth_cm;
void main() {
    Particle p = particles[gl_InstanceIndex];
    float angle = p.center_angle.w * frame.source.w;
    float c=cos(angle), s=sin(angle);
    vec3 v=source_position;
    v.xy += vec2(p.pivot_dynamic_speed.x,-p.pivot_dynamic_speed.y);
    v *= p.scale_alpha.xyz;
    vec3 center=(frame.world_to_view*vec4(p.center_angle.xyz,1.0)).xyz;
    vec3 view=center+vec3(c*v.x-s*v.y,s*v.x+c*v.y,v.z);
    gl_Position=frame.projection*vec4(view,1.0);
    uv=source_uv;
    particle=vec4(c,s,p.scale_alpha.w,p.pivot_dynamic_speed.z);
    speed_cm_s=p.pivot_dynamic_speed.w;
    surface_depth_cm=-view.z*100.0;
}
