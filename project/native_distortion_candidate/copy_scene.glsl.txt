#version 450
layout(local_size_x=8,local_size_y=8,local_size_z=1) in;
layout(set=0,binding=0,rgba16f) uniform readonly image2D source_scene;
layout(set=0,binding=1,rgba16f) uniform writeonly image2D scene_snapshot;
layout(push_constant,std430) uniform Push { ivec2 size; int audit_mode; int unused; } p;
void main() {
    ivec2 pixel=ivec2(gl_GlobalInvocationID.xy);
    if(any(greaterThanEqual(pixel,p.size)))return;
    imageStore(scene_snapshot,pixel,imageLoad(source_scene,pixel));
}
