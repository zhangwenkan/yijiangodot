#version 450
layout(set=0,binding=1,std140) uniform Frame {
 mat4 projection; mat4 inverse_projection; mat4 world_to_view; mat4 camera;
 vec4 size; vec4 material; vec4 atlas_fog;
 vec4 hf0; vec4 hf1; vec4 hfcolor; vec4 hfdir; vec4 hfdircolor; vec4 hftail;
} frame;

layout(location=0) in vec2 uv;
layout(location=1) flat in vec4 tint;
layout(location=2) in float receiver_z;
layout(location=3) in vec4 vertex_fog;
layout(location=4) in float particle_frame;
layout(set=0,binding=2) uniform sampler2D source_texture;
layout(set=0,binding=3) uniform sampler2D opaque_depth;
layout(location=0) out vec4 output_color;
float radial(vec2 p){float r=1.0-2.0*length(p-vec2(.5));float e=r*2.3329999446868896;e=e*e;float q=exp2(e*1.442695140838623);return r>=0.0&&abs(r)>9.999999747378752e-6?1.0-1.0/q:0.0;}
void main(){
 float index=floor(particle_frame);
 vec2 source_uv=(uv+vec2(mod(index,frame.atlas_fog.x),floor(index/frame.atlas_fog.x)))/frame.atlas_fog.xy;
 vec2 suv=gl_FragCoord.xy/frame.size.xy;
 float device_z=textureLod(opaque_depth,suv,0).r;
 // Original unlit translucency keeps depth testing, but never writes opaque depth.
 if(gl_FragCoord.z<device_z)discard;
 vec3 emission;float alpha;
 if(frame.material.x<1.5){
  vec4 texel=texture(source_texture,source_uv,frame.material.w);
  emission=max(texel.rgb*tint.rgb,vec3(0));
  alpha=clamp((frame.material.x<.5?clamp(texel.a,0.0,1.0):texel.a)*tint.a,0.0,1.0);
  if(frame.material.x<.5){
   vec4 v=frame.inverse_projection*vec4(suv*2.0-1.0,device_z,1);
   alpha*=clamp((-v.z/v.w+receiver_z)*100.0/frame.material.y,0.0,1.0);
  }
 }else{
  float mask=radial(source_uv);emission=max(tint.rgb*frame.material.z,vec3(0))*mask;
  alpha=clamp(mask*tint.a,0.0,1.0);
 }
 // Exact diagnostic branches of the original material, inactive in normal mode0.
 if(frame.atlas_fog.w>0.5&&frame.atlas_fog.w<1.5){emission=vec3(fract(source_uv),0);alpha=1.0;}
 if(frame.atlas_fog.w>1.5&&frame.atlas_fog.w<2.5){emission=vec3(alpha);alpha=1.0;}
 if(frame.atlas_fog.w>2.5){emission/=50.0;alpha=1.0;}
 // Source blend-mix fog: HDR source emission * T + premultiplied inscattering.
 emission=emission*vertex_fog.a+vertex_fog.rgb;
 output_color=vec4(emission,alpha);
}
