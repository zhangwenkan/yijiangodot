#version 450
// Copy-rasterizes the existing source mesh. No generated tessellation or geometry.
layout(location=0) in vec3 source_position;
layout(location=1) in vec3 source_normal;
layout(location=2) in vec2 source_uv;
layout(set=0,binding=0,std140) uniform WaterFrame {
    mat4 projection; mat4 inverse_projection; mat4 world_to_view; mat4 view_to_world;
    vec4 render_output_size; mat4 model;
    vec4 p0; vec4 p1; vec4 p2; vec4 p3; vec4 p4; vec4 p5;
} f;
// Godot 4.7.2 float build RenderSceneDataRD::UBO; source TIME uses this same field.
// 2832 byte offset proved from retained struct and MAX_RENDER_VIEWS=2.
layout(set=0,binding=8,std140) uniform SceneClock { vec4 preceding[177]; vec4 time_taa; } clock_data;
layout(location=0) out vec2 uv;
layout(location=1) out vec3 view_position;
layout(location=2) out vec3 flat_view_normal;
void main() {
    float t=clock_data.time_taa.x;
    vec4 world=f.model*vec4(source_position,1.0);
    vec2 phase=(world.xz*100.0/max(f.p4.y*f.p0.x,0.000001)+vec2(t*f.p4.z))*6.28318530718;
    world.y+=dot(sin(phase)*0.5+vec2(0.5),vec2(1.0))*f.p4.w*0.01+f.p5.x;
    vec4 view=f.world_to_view*world;
    gl_Position=f.projection*view;
    uv=source_uv;
    view_position=view.xyz;
    flat_view_normal=normalize(transpose(inverse(mat3(f.world_to_view*f.model)))*source_normal);
}
